local function checkAlly(p)
	if p==L or (p.Team and p.Team==L.Team) or (not p.Neutral and not L.Neutral and p.TeamColor==L.TeamColor) then return true end
	for _,a in ipairs({"Team","team","Side","Faction"}) do
		if (p:GetAttribute(a)~=nil and p:GetAttribute(a)==L:GetAttribute(a)) or (p.Character and L.Character and p.Character:GetAttribute(a)~=nil and p.Character:GetAttribute(a)==L.Character:GetAttribute(a)) then return true end
		local v1,v2=p:FindFirstChild(a,true),L:FindFirstChild(a,true)
		if v1 and v2 and v1:IsA("ValueBase") and v1.Value==v2.Value then return true end
	end
	return p.Character and L.Character and p.Character.Parent~=workspace and p.Character.Parent==L.Character.Parent
end

local t1,t2,t3=tab("🎯 Combat",1),tab("👁 Visuals",2),tab("⛄ Player",3)

local function upd()
	for _,p in ipairs(P:GetPlayers()) do
		allies[p]=checkAlly(p)
		if p~=L and p.Character then
			local al=allies[p] local hl=p.Character:FindFirstChild("StrikeHL")
			if c.esp and not al then if not hl then mk("Highlight",p.Character,{Name="StrikeHL",FillColor=Color3.fromRGB(65,180,255),OutlineColor=Color3.new(1,1,1),FillTransparency=0.45,DepthMode=Enum.HighlightDepthMode.AlwaysOnTop}) end
			elseif hl then hl:Destroy() end
			local hd=p.Character:FindFirstChild("Head") local tg=hd and hd:FindFirstChild("StrikeTag")
			if c.nm and hd and not al then
				if not tg then tg=mk("BillboardGui",hd,{Name="StrikeTag",Size=UDim2.new(0,120,0,30),StudsOffset=Vector3.new(0,2.5,0),AlwaysOnTop=true}) mk("TextLabel",tg,{Name="I",Size=UDim2.new(1,0,1,0),BackgroundTransparency=1,Text=p.Name,TextColor3=Color3.fromRGB(220,245,255),TextStrokeTransparency=0.3,Font=Enum.Font.GothamBold,TextSize=12}) end
			elseif tg then tg:Destroy() end
		end
	end
end

local function tgt(cam)
	local b,d,cn=nil,c.fov and c.rad or 9999,Vector2.new(cam.ViewportSize.X*0.5,cam.ViewportSize.Y*0.5)
	for _,p in ipairs(P:GetPlayers()) do
		if not allies[p] and p.Character then
			local h=p.Character:FindFirstChildOfClass("Humanoid")
			local pt=p.Character:FindFirstChild(c.pt) or p.Character:FindFirstChild("HumanoidRootPart")
			if h and h.Health>0 and pt then
				local sp,v=cam:WorldToViewportPoint(pt.Position)
				if v then local ds=(Vector2.new(sp.X,sp.Y)-cn).Magnitude if ds<d then d,b=ds,pt end end
			end
		end
	end
	return b
end

	if i.KeyCode==Enum.KeyCode.RightShift then open=not open
		if open then m.Visible=true T:Create(sc,TweenInfo.new(0.28,Enum.EasingStyle.Back,Enum.EasingDirection.Out),{Scale=1}):Play()
		else local tw=T:Create(sc,TweenInfo.new(0.22,Enum.EasingStyle.Quad,Enum.EasingDirection.In),{Scale=0}) tw:Play() tw.Completed:Once(function() if not open then m.Visible=false end end) end
	elseif i.UserInputType==Enum.UserInputType.MouseButton2 then hold=true upd()
	elseif i.KeyCode==Enum.KeyCode.Space and c.inf then local h=L.Character and L.Character:FindFirstChildOfClass("Humanoid") if h then h:ChangeState(Enum.HumanoidStateType.Jumping) end end
end)
